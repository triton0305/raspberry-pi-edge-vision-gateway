cmake_minimum_required(VERSION 3.16)
project(pi_gateway VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(GATEWAY_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(GATEWAY_SANITIZER_SET "address,undefined" CACHE STRING "GCC/Clang sanitizer list")
if(GATEWAY_SANITIZERS)
  add_compile_options(-fsanitize=${GATEWAY_SANITIZER_SET} -fno-omit-frame-pointer -fno-sanitize-recover=all)
  add_link_options(-fsanitize=${GATEWAY_SANITIZER_SET})
endif()
find_package(Threads REQUIRED)
add_executable(pi_gateway src/gateway.c src/json.c)
target_compile_definitions(pi_gateway PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(pi_gateway PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(pi_gateway PRIVATE Threads::Threads)
include(CTest)
if(BUILD_TESTING)
  add_executable(json_tests tests/json_test.c src/json.c)
  target_include_directories(json_tests PRIVATE src)
  target_compile_options(json_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME json_parser COMMAND json_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:pi_gateway>)
  add_test(NAME recovery COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/recovery.py $<TARGET_FILE:pi_gateway>)
  add_test(NAME session_sync COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/session_sync.py $<TARGET_FILE:pi_gateway>)
  set_tests_properties(integration recovery session_sync PROPERTIES TIMEOUT 45)
endif()
install(TARGETS pi_gateway RUNTIME DESTINATION bin)
